terraform {
  required_version = ">= 1.5.0"
  required_providers {
    okta = {
      source  = "okta/okta"
      version = "~> 4.6.0"
    }
  }
}

provider "okta" {
  org_name  = var.okta_org_name
  base_url  = var.okta_base_url
  api_token = var.okta_api_token
}

# --- GROUPS ---
resource "okta_group" "secops_admins" {
  name        = "SecOps-Administrators"
  description = "Group for Security Operations Team with elevated access"
}

resource "okta_group" "all_employees" {
  name        = "All-Employees"
  description = "Base group assigned to all onboarded users"
}

# --- PASSWORD POLICY ---
resource "okta_policy_password" "strict_password_policy" {
  name                   = "Strict-Enterprise-Password-Policy"
  status                 = "ACTIVE"
  description            = "Enforces 16+ char passwords with mandatory MFA"
  password_min_length    = 16
  password_min_uppercase = 1
  password_min_lowercase = 1
  password_min_number    = 1
  password_min_symbol    = 1
  password_history_count = 10
  
  groups_included = [okta_group.all_employees.id]
}

# --- OIDC APP REGISTRATION ---
resource "okta_app_oauth" "internal_portal" {
  label                      = "Internal SecOps Developer Portal"
  type                       = "web"
  grant_types                = ["authorization_code", "refresh_token"]
  redirect_uris              = ["https://portal.internal.domain/login/callback"]
  post_logout_redirect_uris = ["https://portal.internal.domain/logout"]
  response_types             = ["code"]
  token_endpoint_auth_method = "client_secret_basic"

  groups = [okta_group.secops_admins.id]
}
